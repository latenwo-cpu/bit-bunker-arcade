rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Admin = signed in with email+password (admin.html).
    function isAdmin() {
      return request.auth != null
          && request.auth.token.firebase.sign_in_provider == 'password';
    }

    // Online multiplayer rooms (room-code trust model, no Firebase Auth).
    match /rooms/{roomCode} {
      allow read, write: if true;

      // Chat: length-limited, must carry sender + uid, cannot be edited later.
      match /chat/{msgId} {
        allow read: if true;
        allow create: if request.resource.data.text is string
                      && request.resource.data.text.size() > 0
                      && request.resource.data.text.size() <= 300
                      && request.resource.data.sender is string
                      && request.resource.data.uid is string;
        allow delete: if true;
      }

      match /spectators/{uid} {
        allow read, write: if true;
      }
    }

    // Leaderboards: score must be a sane number and the doc must belong to the uid in its path.
    match /leaderboards/{gameId}/scores/{uid} {
      allow read: if true;
      allow create, update: if request.resource.data.uid == uid
                            && request.resource.data.score is number
                            && request.resource.data.score >= 0
                            && request.resource.data.score < 10000000000
                            && request.resource.data.name is string
                            && request.resource.data.name.size() <= 40;
      allow delete: if true;   // needed for "Delete my online data"
    }

    // Player directory + friends.
    match /players/{code} {
      allow read: if true;
      allow create, update: if request.resource.data.code == code
                            && request.resource.data.name is string
                            && request.resource.data.name.size() <= 40;
      allow delete: if true;   // needed for "Delete my online data"
      match /friends/{friendUid} {
        allow read, delete: if true;
        allow create, update: if request.resource.data.keys().size() < 10;
      }
    }

    // Tournaments (shape varies, kept open).
    match /tournaments/{tournamentId} {
      allow read, write: if true;
    }

    // Analytics events: anyone can create (size-capped), only admin can read.
    match /events/{eventId} {
      allow create: if request.resource.data.keys().size() < 80;
      allow read, update, delete: if isAdmin();
    }

    // Remote ad config: public read, admin write.
    match /config/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Cloud Save: owner only.
    match /cloudProfiles/{uid} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }

    // Chat reports: strict shape; only admin can read them.
    match /reports/{reportId} {
      allow create: if request.resource.data.type == 'chat'
                    && request.resource.data.text is string
                    && request.resource.data.text.size() <= 300
                    && request.resource.data.keys().hasOnly(
                         ['type','reporter','room','msgId','uid','sender','text','ts']);
      allow read, update, delete: if isAdmin();
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
